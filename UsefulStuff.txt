function drawLeak(xBase, yBase) {
  if (frameCount % 5 === 0) {
    droplets.push({ x: random(xBase, xBase + 200), y: yBase, speed: random(1, 3), duration: 0 });
  }

  for (let i = droplets.length - 1; i >= 0; i--) {
    let d = droplets[i];
    fill(100, 100, 255, 180);
    noStroke();
    ellipse(d.x, d.y, 5, 8);
    d.y += d.duration < 6 ? -d.speed : d.speed;
    d.x += d.x > 225 ? d.speed / 2 : -d.speed / 2;
    d.duration++;
    if (d.y > height) droplets.splice(i, 1);
  }
}

function drawControlArrows() {
  const canvasRect = canvas.elt.getBoundingClientRect();

  const drawArrowToSlider = (sliderId, label, color) => {
    const bounds = select(sliderId).elt.getBoundingClientRect();
    const targetX = bounds.left + bounds.width / 2 - canvasRect.left;
    const targetY = bounds.top + bounds.height / 2 - canvasRect.top;

    const sourceX = width * 0.44;
    const sourceY = height * 0.2;

    stroke(color);
    strokeWeight(2);
    //drawingContext.setLineDash([5, 3]); 
    line(sourceX, sourceY, targetX, targetY);
    drawingContext.setLineDash([]);

    // Arrowhead
    const angle = atan2(targetY - sourceY, targetX - sourceX);
    const arrowSize = 10;
    push();
    translate(targetX, targetY);
    rotate(angle);
    fill(color);
    noStroke();
    triangle(0, 0, -arrowSize, arrowSize / 2, -arrowSize, -arrowSize / 2);
    pop();

    // Label
    noStroke();
    fill(color);
    textSize(14);
    textAlign(CENTER, CENTER);
    text(label, sourceX - 20, sourceY);
  };
